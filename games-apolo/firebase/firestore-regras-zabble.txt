    // Ranking do Tiny Zabble Pixel Pop
    match /bubble_scores/{id} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['name','score','t'])
        && request.resource.data.name is string
        && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 8
        && request.resource.data.score is int
        && request.resource.data.score >= 0 && request.resource.data.score <= 500000
        && request.resource.data.t is int;
      allow update, delete: if false;
    }
